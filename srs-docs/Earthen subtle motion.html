Settle on load

Use on the login screen, the home hero, and checkout headings. Elements rise 14 px and fade in one after another.

Reveal on scroll

Use on product grids and content blocks, once each, as they come into view.

Respond to actions

Fields and buttons react to focus, hover, press, and success. Everything switches off for people who prefer reduced motion.

/* globals.css */
@keyframes settle { from { opacity: 0; transform: translateY(14px) } }
.settle { animation: settle .8s cubic-bezier(.2,.7,.2,1) backwards;
          animation-delay: calc(var(--i, 0) * 90ms + 80ms) }
.reveal { opacity: 0; transform: translateY(16px);
          transition: opacity .8s cubic-bezier(.2,.7,.2,1), transform .8s cubic-bezier(.2,.7,.2,1);
          transition-delay: calc(var(--i, 0) * 80ms) }
.reveal.in { opacity: 1; transform: none }
@media (prefers-reduced-motion: reduce) {
  .settle { animation: none }
  .reveal { opacity: 1; transform: none; transition: none }
}

/* usage */
<h1 class="settle" style="--i:0">Welcome back</h1>
<p  class="settle" style="--i:1">Sign in to continue</p>
<div class="reveal" style="--i:2">...product card...</div>

/* one small client script (a React hook in Next.js) */
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target) }
}), { threshold: .15 });
document.querySelectorAll('.reveal').forEach(el => io.observe(el));
